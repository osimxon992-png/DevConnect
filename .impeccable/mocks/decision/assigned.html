Finally replaced our hand-rolled retry loop with a tiny helper. Exponential backoff, jitter, and it respects AbortSignal.

export async function retry(fn, { tries = 5, signal } = {}) {
  for (let i = 0; i < tries; i++) {
    try { return await fn() } catch (e) {
      if (signal?.aborted || i === tries - 1) throw e
      await sleep(2 ** i * 100 + Math.random() * 100)
    }
  }
}
